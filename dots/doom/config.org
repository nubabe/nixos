#+begin_src emacs-lisp
(defun my/agenda-skip-tagged (skip-tag &optional exempt-tags)
  "Skip subtree if tagged SKIP-TAG, unless it has any tag in EXEMPT-TAGS."
  (let ((tags (org-get-tags-at)))
    (when (and (member skip-tag tags)
               (not (seq-intersection exempt-tags tags)))
      (save-excursion (org-end-of-subtree t)))))

(use-package org
  :custom
  (org-agenda-custom-commands
   '(
     ("c" "Calendar" agenda ""
      (
       (org-agenda-skip-function
        '(or (my/agenda-skip-tagged "skip" '("@nuyan" "@alle" "@all" "@me"))
             (org-agenda-skip-entry-if 'nottodo 'todo)))))
     ("t" "todo" todo "TODO"
      ((org-agenda-skip-function
        '(or (my/agenda-skip-tagged "skip" '("@nuyan" "@alle" "@all" "@me"))
             (org-agenda-skip-entry-if 'timestamp)))
       (org-agenda-overriding-header "TODOs")))
     ("e" "exams" todo "EXAM"
      ((org-agenda-overriding-header "Exams")
       (org-agenda-sorting-strategy '((todo time-up)))
       (org-agenda-prefix-format '((todo . " %s ")))))
     ("g" "prayer" todo "PRAY"
      ((org-agenda-overriding-header "Prayer")))))
  )

(use-package org-modern
  :after org
  :custom
  (org-modern-star 'replace)
  :config
  (defun my/org-modern-set-todo-faces ()
  (let ((fg (doom-color 'bg)))
    (setq org-modern-todo-faces
          `(("TODO"        :background ,(doom-color 'green)   :foreground ,fg)
            ("APPOINTMENT" :background ,(doom-color 'violet)  :foreground ,fg)
            ("CLASS"       :background ,(doom-color 'magenta) :foreground ,fg)
            ("EXAM"        :background ,(doom-color 'yellow)  :foreground ,fg)
            ("WAIT"        :background ,(doom-color 'grey)    :foreground ,fg))))
  (when (bound-and-true-p global-org-modern-mode)
    (global-org-modern-mode -1)
    (global-org-modern-mode 1)))

  (add-hook 'doom-load-theme-hook #'my/org-modern-set-todo-faces)
  (my/org-modern-set-todo-faces)
  (global-org-modern-mode 1))
